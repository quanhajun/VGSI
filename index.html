Add-Type -AssemblyName System.IO.Compression.FileSystem $zipPath = "c:\Users\ADMIN\Desktop\PROJECT CONDITION\PROJECT CONDITION1.xlsx" $zip = [System.IO.Compression.ZipFile]::OpenRead($zipPath) $sharedStrings = @() $ssEntry = $zip.GetEntry("xl/sharedStrings.xml") if ($ssEntry) { $stream = $ssEntry.Open() $reader = New-Object System.IO.StreamReader($stream) $xml = [xml]$reader.ReadToEnd() $reader.Close() $stream.Close() foreach ($si in $xml.sst.si) { if ($si.t) { $sharedStrings += $si.t.InnerText } elseif ($si.r) { $tArr = @($si.r | ForEach-Object { $_.t.InnerText }) $sharedStrings += ($tArr -join "") } else { $sharedStrings += "" } } } Write-Output "Shared Strings Count: $($sharedStrings.Count)" # Read workbook.xml to get sheet names $wbEntry = $zip.GetEntry("xl/workbook.xml") $stream = $wbEntry.Open() $reader = New-Object System.IO.StreamReader($stream) $wbXml = [xml]$reader.ReadToEnd() $reader.Close() $stream.Close() Write-Output "`nSheets in Workbook:" foreach ($sheet in $wbXml.workbook.sheets.sheet) { Write-Output "Sheet: $($sheet.name), id: $($sheet.sheetId), r:id: $($sheet.id)" } $zip.Dispose()
